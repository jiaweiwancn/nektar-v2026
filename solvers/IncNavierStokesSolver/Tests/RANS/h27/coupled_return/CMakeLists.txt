cmake_minimum_required(VERSION 3.16)
project(WJH27CoupledReturn CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Nektar++ REQUIRED)
find_package(Eigen3 REQUIRED NO_MODULE)
if(NOT H27_PINNED_SOURCE OR NOT H27_PINNED_BUILD)
    message(FATAL_ERROR "Explicit immutable source/build paths are required")
endif()
include("${H27_PINNED_SOURCE}/solvers/IncNavierStokesSolver/Tests/RANS/h27/source_identity.cmake")
wj_h27_source_identity("${H27_PINNED_SOURCE}" h27_identity)
if(NOT h27_identity STREQUAL "56c35390fdf092180913f60c3ae348ca10cdcdf4e6b889c3afd589eddb511acf")
    message(FATAL_ERROR "This focused regression requires exact source19")
endif()
add_definitions(${NEKTAR++_DEFINITIONS} -DENABLE_NEKTAR_EXCEPTIONS)
add_executable(WJH27CoupledReturnProbe ../coupled_return_probe.cpp)
target_compile_definitions(WJH27CoupledReturnProbe PRIVATE WJ_H27_SOURCE_ID="${h27_identity}")
target_include_directories(WJH27CoupledReturnProbe PRIVATE "${H27_PINNED_SOURCE}/library" "${H27_PINNED_SOURCE}/solvers")
target_include_directories(WJH27CoupledReturnProbe SYSTEM PRIVATE ${NEKTAR++_INCLUDE_DIRS} ${NEKTAR++_TP_INCLUDE_DIRS})
target_compile_options(WJH27CoupledReturnProbe PRIVATE -O3 -ffp-contract=off -fno-fast-math -Wall -Wextra)
target_link_libraries(WJH27CoupledReturnProbe PRIVATE
    "${H27_PINNED_BUILD}/libWJNativeMixedFlow.a"
    "${H27_PINNED_BUILD}/h26-kernel/libWJNativeFlow.a"
    Eigen3::Eigen ${NEKTAR++_LIBRARIES} ${NEKTAR++_TP_LIBRARIES})
